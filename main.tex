% ---- Projects ----
\section*{Projects}
\begin{itemize}[leftmargin=*, itemsep=7pt, parsep=0pt, topsep=3pt]

    \item \textbf{CARDD -- Car Damage Detection with YOLO} \hfill \cvmeta{Jun 2026}\\
    Trained and benchmarked YOLO11-L against YOLOv8-L on the CarDD dataset (4,000 images, 8,740 annotated boxes across 6 damage classes: scratches, dents, cracks, broken lamps, shattered glass, flat tires). YOLO11-L achieved \textbf{mAP@50--95 of 0.557}.\\
    \textbf{Tech:} Python, YOLO11, YOLOv8, PyTorch, Object Detection \quad \href{https://github.com/Aditi-Adri/CARDD---Car-Damage-Detection-with-YOLO}{GitHub}

    \item \textbf{Pet Image Segmentation \& Breed Classification (U-Net)} \hfill \cvmeta{May 2026}\\
    Deep learning project using U-Net and Attention U-Net architectures for binary pet image segmentation and 37-breed classification on the Oxford-IIIT Pet Dataset. Achieved \textbf{$\sim$88\% IoU} with a pretrained ResNet-34 encoder. Implemented data augmentation, combined loss functions, and ablation studies.\\
    \textbf{Tech:} Python, PyTorch, Jupyter Notebook \quad \href{https://github.com/Aditi-Adri/Image-Processing-Project---pet-segmentation-classification-unet-CSE428}{GitHub}

    \item \textbf{Humanitarian Classification of Disaster Tweets} \hfill \cvmeta{Sep 2026}\\
    Benchmarked ten models on the HumAID dataset (76,484 tweets, 19 disasters, 10 humanitarian categories with a 59.6:1 class imbalance). \textbf{BERT reached 0.768 macro F1}; TF-IDF logistic regression followed at 0.715, outperforming every recurrent model tested.\\
    \textbf{Tech:} Python, BERT, TF-IDF, Scikit-Learn, NLP \quad \href{https://github.com/Aditi-Adri/Humanitarian-Category-Classification-of-Disaster-Tweets}{GitHub}

    \item \textbf{SaveLife -- SOS Emergency Response App} \hfill \cvmeta{Jul 2026}\\
    Emergency response platform with one-tap SOS alerts and GPS geolocation, a live responder dashboard, and role-based JWT authentication. React/Vite frontend, Node.js/Express API, PostgreSQL, deployed on Render.\\
    \textbf{Tech:} React, Vite, Node.js, Express, PostgreSQL, JWT \quad \href{https://github.com/Aditi-Adri/SaveLife}{GitHub}

    \item \textbf{Roaming Sonic -- Travel \& Tour Management Platform} \hfill \cvmeta{Jan 2026}\\
    Full-stack MERN web application for Bangladesh tourism featuring tour packages, hotel bookings, bus tickets, guide services, AI chatbot, and referral rewards system. Live-deployed on Vercel.\\
    \textbf{Tech:} MongoDB, Express.js, React.js, Node.js \quad \href{https://github.com/Shiva-Prasad-Sarkar/Roaming-Sonic}{GitHub} $|$ \href{https://roaming-sonic.vercel.app/}{Live Demo}

    \item \textbf{Metadata Journal System} \hfill \cvmeta{Jan 2026}\\
    OS-level project implementing file metadata management and journaling mechanisms for maintaining filesystem consistency.\\
    \textbf{Tech:} C \quad \href{https://github.com/Aditi-Adri/CSE321-metadata-journal}{GitHub}

\end{itemize}
